Drop the media attribute of the first style element only once its styles are
wrapped in @media, when a later style element is merged into it.

===

<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100">
    <style media="print">.no-print{display:none}</style>
    <style media="(prefers-color-scheme: dark)">.a{fill:white}</style>
    <rect width="100" height="100" class="no-print a"/>
</svg>

@@@

<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100">
    <style>
        @media print{.no-print{display:none}}@media (prefers-color-scheme: dark){.a{fill:white}}
    </style>
    <rect width="100" height="100" class="no-print a"/>
</svg>
