Don't inline declarations that a dynamic pseudo-class rule matching the same
element also sets, including shorthand and longhand overlaps, as the inline
style would take precedence over it. Rules that lose to a retained rule stay
retained too.

See: https://github.com/svg/svgo/issues/2067

===

<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 50 50">
  <style>
    circle {
      fill: blue;
    }
    circle:active {
      fill: yellow;
    }
    #a {
      stroke: red;
    }
    #a:hover {
      stroke: green;
    }
    ellipse.foo {
      fill: blue;
      stroke: red;
    }
    ellipse:hover {
      fill: yellow;
    }
    ellipse {
      stroke: green;
    }
    text {
      font: 12px serif;
    }
    text:hover {
      font-size: 24px;
    }
    rect {
      fill: purple;
    }
  </style>
  <circle cx="5" cy="10" r="1"/>
  <path id="a" d="M2 2h10"/>
  <ellipse class="foo" cx="5" cy="20" rx="2" ry="1"/>
  <text y="30">a</text>
  <rect width="5" height="5"/>
</svg>
