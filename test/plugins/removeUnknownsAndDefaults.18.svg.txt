Keeps pathLength on basic shapes, where it is a valid attribute.

See: https://github.com/svg/svgo/issues/2310

===

<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100">
  <circle cx="20" cy="20" r="10" pathLength="36"/>
  <ellipse cx="50" cy="20" rx="10" ry="5" pathLength="36"/>
  <line x1="0" y1="40" x2="100" y2="40" pathLength="36"/>
  <polygon points="10,50 20,70 0,70" pathLength="36"/>
  <polyline points="30,50 40,70 50,50" pathLength="36"/>
  <rect x="60" y="50" width="20" height="20" pathLength="36"/>
  <path d="M0 90h100" pathLength="36"/>
  <circle cx="20" cy="20" r="10" foo="bar"/>
</svg>
