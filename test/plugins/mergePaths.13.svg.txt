Do not merge paths when the `marker` shorthand applies, because merged paths lose their ends and markers are rendered incorrectly.

===

<svg width="100" height="100">
    <defs>
        <style>
            .a {marker: url(#arrowhead);}
        </style>
        <marker id="arrowhead" markerWidth="10" markerHeight="10" refX="6" refY="3">
            <path d="M 0,0 l 6,3 l -6,3" stroke="black"/>
        </marker>
    </defs>
    <path d="M 10,60 h60" stroke="black" class="a"/>
    <path d="M 10,70 h60" stroke="black" class="a"/>
</svg>

@@@
