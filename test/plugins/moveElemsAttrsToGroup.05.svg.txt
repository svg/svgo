Do not move transforms between elements with different origins

===

<svg xmlns="http://www.w3.org/2000/svg">
    <g transform-origin="50 50">
        <g transform="scale(2)">
            <path d="M10 10h5v5h-5z"/>
        </g>
        <g transform="scale(2)">
            <path d="M20 20h5v5h-5z"/>
        </g>
    </g>
    <g>
        <g transform="scale(2)" transform-origin="50 50">
            <path d="M40 40h20v20H40z"/>
        </g>
        <g transform="scale(2)" transform-origin="50 50">
            <path d="M10 10h5v5h-5z"/>
        </g>
    </g>
</svg>

@@@
